```python
import pandas as pd

def transform(source: pd.DataFrame) -> pd.DataFrame:
    df = source.copy()
    date_str = df['CST'].str.split(' ').str[-1]
    dt = pd.to_datetime(date_str, format='%m/%d/%Y')
    df['dt_strata'] = dt.dt.strftime('%m/%d/%Y')
    df['dow'] = dt.dt.dayofweek + 1
    rename_dict = {f'{i}:00': f'pct_hourly_{i:02d}00' for i in range(1, 24)}
    if '24:00:00' in df.columns:
        rename_dict['24:00:00'] = 'pct_hourly_2400'
    elif '24:00' in df.columns:
        rename_dict['24:00'] = 'pct_hourly_2400'
    df = df.rename(columns=rename_dict)
    df['pct_hourly_2500'] = 0
    target_cols = ['dt_strata', 'dow'] + [f'pct_hourly_{i:02d}00' for i in range(1, 26)]
    return df[target_cols]
```
