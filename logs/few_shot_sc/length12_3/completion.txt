def transform(source: pd.DataFrame) -> pd.DataFrame:

    # Parse timestamp
    df['timestamp_dt'] = pd.to_datetime(df['timestamp'], format='%Y-%m-%d %H:%M:%S')

    def get_season(m):
        if m in [12, 1, 2]:
            return 'Winter'
        elif m in [3, 4, 5]:
            return 'Spring'
        elif m in [6, 7, 8]:
            return 'Summer'
        else:
            return 'Fall'
    df['season'] = month.apply(get_season)
